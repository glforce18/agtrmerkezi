🎮 SUNUCU KONTROL ÖZELLİKLERİ (10 MADDE)
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━

1. Sunucu Başlatma (Start)
   - Tek tıkla sunucuyu başlat
   - Screen session ile arka planda çalıştır
   - API: POST /api/servers/{id}/start

2. Sunucu Durdurma (Stop)
   - Tek tıkla sunucuyu durdur
   - Güvenli kapatma (oyuncular uyarılsın)
   - API: POST /api/servers/{id}/stop

3. Sunucu Yeniden Başlatma (Restart)
   - Otomatik restart
   - Oyunculara countdown göster
   - API: POST /api/servers/{id}/restart

4. Sunucu Durum Kontrolü (Status)
   - Anlık online/offline durumu
   - Yeşil/kırmızı gösterge
   - Real-time güncelleme (WebSocket)
   - API: GET /api/servers/{id}/status

5. Otomatik Yeniden Başlatma
   - Sunucu çökerse otomatik restart
   - Healthcheck her 60 saniye
   - API: POST /api/servers/{id}/auto-restart/enable

6. Zamanlı Restart
   - Belirli saatte otomatik restart
   - Cron job ile (örn: her gün 05:00)
   - API: POST /api/servers/{id}/scheduled-restart

7. Sunucu Kopyalama
   - Mevcut sunucudan yeni sunucu oluştur
   - Tüm ayarları kopyala
   - API: POST /api/servers/{id}/clone

8. Sunucu Silme
   - Sunucuyu tamamen sil
   - Onaylı silme (2-step confirmation)
   - API: DELETE /api/servers/{id}

9. Sunucu Dondurma (Freeze)
   - Geçici olarak dondur
   - Ücret kesilmesin
   - API: POST /api/servers/{id}/freeze

10. Sunucu Taşıma
    - Farklı user'a sunucu devret
    - Owner değiştirme
    - API: POST /api/servers/{id}/transfer

# ID Oluşturma
def generate_server_code():
    """
    AGTR-2026-00001 formatında ID oluştur
    """
    year = datetime.now().year
    
    # Son server numarasını al
    last_server = db.execute(
        select(GameServer)
        .where(GameServer.server_code.like(f"AGTR-{year}-%"))
        .order_by(GameServer.id.desc())
    ).first()
    
    if last_server:
        last_num = int(last_server.server_code.split("-")[-1])
        new_num = last_num + 1
    else:
        new_num = 1
    
    return f"AGTR-{year}-{new_num:05d}"

async def create_server(user_id: int, package_id: int, server_name: str):
    """
    Sunucu oluştur (tam kontrol)
    """
    # 1. Kullanıcı var mı?
    user = await db.get(User, user_id)
    if not user:
        raise ValueError("Kullanıcı bulunamadı")
    
    # 2. Paket var mı?
    package = await db.get(Package, package_id)
    if not package:
        raise ValueError("Paket bulunamadı")
    
    # 3. Kullanıcının aktif sunucu limiti?
    active_servers = await db.execute(
        select(func.count()).where(
            GameServer.user_id == user_id,
            GameServer.status.in_(['online', 'offline', 'installing'])
        )
    ).scalar()
    
    if active_servers >= user.max_servers:
        raise ValueError(f"Maksimum {user.max_servers} sunucu limiti aşıldı")
    
    # 4. Kullanıcının bakiyesi yeterli mi?
    if user.balance < package.price_monthly:
        raise ValueError("Yetersiz bakiye")
    
    # 5. Aynı isimde sunucu var mı?
    existing = await db.execute(
        select(GameServer).where(
            GameServer.server_name == server_name,
            GameServer.user_id == user_id
        )
    ).first()
    
    if existing:
        raise ValueError("Bu isimde bir sunucunuz zaten var")
    
    # 6. Boş port bul
    port = await find_available_port()
    if not port:
        raise ValueError("Uygun port bulunamadı")
    
    # 7. Disk alanı yeterli mi?
    total_disk = await get_total_disk_usage()
    if total_disk + package.disk_space > MAX_TOTAL_DISK:
        raise ValueError("Sunucu disk alanı dolu")
    
    # ✅ OLUŞTUR
    server = GameServer(
        uuid=str(uuid.uuid4()),
        server_code=generate_server_code(),
        user_id=user_id,
        package_id=package_id,
        server_name=server_name,
        port=port,
        max_players=package.max_players,
        status='pending'
    )
    
    db.add(server)
    await db.commit()
    
    # Background task başlat
    install_server_task.delay(server.id)
    
    return server

async def purchase_and_create_server(user_id: int, package_id: int):
    """
    Satın alma + Sunucu oluşturma (atomik)
    """
    async with db.begin():  # Transaction başlat
        try:
            # 1. Bakiyeyi düş
            user = await db.get(User, user_id, with_for_update=True)
            user.balance -= package.price_monthly
            
            # 2. Invoice oluştur
            invoice = Invoice(
                user_id=user_id,
                amount=package.price_monthly,
                description=f"Server purchase - {package.name}"
            )
            db.add(invoice)
            
            # 3. Sunucu oluştur
            server = await create_server(user_id, package_id, server_name)
            
            # ✅ Commit (hepsi başarılı)
            await db.commit()
            
            return server
            
        except Exception as e:
            # ❌ Rollback (herhangi biri başarısız)
            await db.rollback()
            raise

# app/tasks/server_installation.py
from celery import Task
import subprocess
import os

@celery_app.task(bind=True, max_retries=3)
def install_server_task(self, server_id: int):
    """
    Sunucu kurulum görevi
    """
    try:
        # Server bilgilerini al
        server = db.get(GameServer, server_id)
        server.status = "installing"
        server.installation_progress = 0
        db.commit()
        
        base_path = f"/home/gameservers/server_{server_id}"
        
        # ADIM 1: Dizin oluştur
        logger.info(f"Creating directory: {base_path}")
        os.makedirs(base_path, exist_ok=True)
        server.installation_progress = 10
        db.commit()
        
        # ADIM 2: SteamCMD ile HLDS indir
        logger.info(f"Downloading HLDS via SteamCMD")
        result = subprocess.run([
            "/home/steamcmd/steamcmd.sh",
            "+force_install_dir", base_path,
            "+login", "anonymous",
            "+app_update", "90", "validate",
            "+quit"
        ], capture_output=True, text=True, timeout=600)
        
        if result.returncode != 0:
            raise Exception(f"SteamCMD failed: {result.stderr}")
        
        server.installation_progress = 40
        db.commit()
        
        # ADIM 3: AMXX kur
        logger.info(f"Installing AMX Mod X")
        subprocess.run([
            "wget", "-O", f"{base_path}/amxx.zip",
            "https://www.amxmodx.org/amxxdrop/1.10/amxmodx-latest-base-linux.tar.gz"
        ], check=True)
        
        subprocess.run([
            "tar", "-xzf", f"{base_path}/amxx.zip",
            "-C", f"{base_path}/valve/"
        ], check=True)
        
        server.installation_progress = 60
        db.commit()
        
        # ADIM 4: Config dosyaları
        logger.info(f"Creating config files")
        create_server_cfg(server_id, server.server_name, server.rcon_password, server.max_players, server.port)
        create_mapcycle_txt(server_id)
        create_users_ini(server_id)
        create_plugins_ini(server_id)
        
        server.installation_progress = 80
        db.commit()
        
        # ADIM 5: Default plugins
        logger.info(f"Installing default plugins")
        copy_default_plugins(server_id)
        
        server.installation_progress = 90
        db.commit()
        
        # ADIM 6: Test başlatma
        logger.info(f"Testing server startup")
        test_result = test_server_startup(server_id, server.port)
        
        if not test_result:
            raise Exception("Server startup test failed")
        
        # ADIM 7: Tamamlandı
        server.status = "offline"
        server.installation_progress = 100
        server.installed_at = datetime.now()
        db.commit()
        
        # Bildirim gönder
        send_notification(
            user_id=server.user_id,
            title="Sunucu Kurulumu Tamamlandı",
            message=f"Sunucunuz {server.server_code} kullanıma hazır!",
            type="success"
        )
        
        logger.info(f"Server {server_id} installation completed successfully")
        
    except Exception as e:
        logger.error(f"Server {server_id} installation failed: {str(e)}")
        
        # Retry (max 3 kez)
        if self.request.retries < self.max_retries:
            raise self.retry(exc=e, countdown=60)  # 60 saniye sonra tekrar dene
        
        # Max retry aşıldı, başarısız olarak işaretle
        server.status = "failed"
        server.installation_error = str(e)
        db.commit()
        
        # Hata bildirimi
        send_notification(
            user_id=server.user_id,
            title="Sunucu Kurulumu Başarısız",
            message=f"Sunucunuz {server.server_code} kurulamadı. Destek ekibiyle iletişime geçin.",
            type="error"
        )
